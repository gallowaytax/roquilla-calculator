<script>
class RoQuillaCalculator {
    constructor() {
        this.minProfitThreshold = 100;
        this.mileageRate = 0.70;
        this.dropoffRate = 0.70;
        this.isPremium = this.checkPremiumAccess();
        this.initializeElements();
        this.bindEvents();
        this.updatePremiumUI();
        this.calculate();

        // Auto-hide overlay if premium
        if (this.isPremium) this.hideUpgradeOverlay();
    }

    checkPremiumAccess() {
        const urlParams = new URLSearchParams(window.location.search);
        if (urlParams.get('premium') === '1') localStorage.setItem('roquillaPremium', '1');
        return localStorage.getItem('roquillaPremium') === '1';
    }

    initializeElements() {
        this.askingFee = document.getElementById('askingFee');
        this.mileageMiles = document.getElementById('mileageMiles');
        this.dropoffMiles = document.getElementById('dropoffMiles');
        this.edocFee = document.getElementById('edocFee');
        this.totalExpensesEl = document.getElementById('totalExpenses');
        this.netProfitEl = document.getElementById('netProfit');
        this.hourlyRateEl = document.getElementById('hourlyRate');
        this.recommendationEl = document.getElementById('recommendation');
        this.recommendationTextEl = document.getElementById('recommendationText');
        this.recommendationReasonEl = document.getElementById('recommendationReason');
        this.showUpgradeBtn = document.getElementById('showUpgradeBtn');
        this.continueFreeLink = document.getElementById('continueFreeLink');
        this.premiumOverlay = document.getElementById('premiumOverlay');
        this.continueFreeBtn = document.getElementById('continueFreeBtn');
        this.closeOverlayBtn = document.getElementById('closeOverlayBtn');
    }

    bindEvents() {
        [this.askingFee, this.mileageMiles, this.dropoffMiles, this.edocFee].forEach(input => input.addEventListener('input', () => this.calculate()));
        this.showUpgradeBtn.addEventListener('click', () => this.showUpgradeOverlay());
        this.continueFreeLink.addEventListener('click', () => this.hideUpgradeOverlay());
        this.continueFreeBtn.addEventListener('click', () => this.hideUpgradeOverlay());
        this.closeOverlayBtn.addEventListener('click', () => this.hideUpgradeOverlay());
        this.premiumOverlay.addEventListener('click', (e) => { if (e.target === this.premiumOverlay) this.hideUpgradeOverlay(); });
    }

    updatePremiumUI() {
        if (this.isPremium) {
            this.edocFee.removeAttribute('readonly');
            this.edocFee.classList.add('premium-unlocked');
        }
    }

    calculate() {
        const askingFee = parseFloat(this.askingFee.value) || 0;
        const mileageCost = (parseFloat(this.mileageMiles.value) || 0) * this.mileageRate;
        const dropoffCost = (parseFloat(this.dropoffMiles.value) || 0) * this.dropoffRate;
        const edocFee = parseFloat(this.edocFee.value) || 0;
        const totalExpenses = mileageCost + dropoffCost + edocFee;
        const netProfit = askingFee - totalExpenses;
        const totalTime = 1;
        const hourlyRate = totalTime > 0 ? netProfit / totalTime : 0;

        this.totalExpensesEl.textContent = `$${totalExpenses.toFixed(2)}`;
        this.netProfitEl.textContent = `$${netProfit.toFixed(2)}`;
        this.hourlyRateEl.textContent = `$${hourlyRate.toFixed(2)}/hr`;

        let recommendation, reason, className;
        if (netProfit < this.minProfitThreshold) { recommendation="DECLINE"; reason=`Below $${this.minProfitThreshold} minimum profit. Current: $${netProfit.toFixed(2)}.`; className='recommendation-decline'; }
        else if (hourlyRate < 25) { recommendation="COUNTER OFFER"; reason=`Hourly rate $${hourlyRate.toFixed(2)}/hr is below $25/hr.`; className='recommendation-counter'; }
        else { recommendation="ACCEPT"; reason=`Profit $${netProfit.toFixed(2)} meets target.`; className='recommendation-accept'; }

        this.recommendationTextEl.textContent = recommendation;
        this.recommendationReasonEl.textContent = reason;
        this.recommendationEl.className = `mt-4 p-4 rounded-lg border-2 ${className}`;
    }

    showUpgradeOverlay() { this.premiumOverlay.classList.add('active'); document.body.style.overflow='hidden'; }
    hideUpgradeOverlay() { this.premiumOverlay.classList.remove('active'); document.body.style.overflow='auto'; }
}

document.addEventListener('DOMContentLoaded', () => { new RoQuillaCalculator(); });
</script>
